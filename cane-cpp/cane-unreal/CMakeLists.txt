cmake_minimum_required(VERSION 3.24)
if(CMAKE_BINARY_DIR STREQUAL CMAKE_SOURCE_DIR OR CMAKE_BINARY_DIR MATCHES "^[Cc]:")
  message(FATAL_ERROR "Use a separate non-C build directory.")
endif()
project(CaneUnrealCore VERSION 0.1.0 LANGUAGES CXX)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The current UE adapter staging target requires MSVC Windows x64.")
endif()
# UE uses the release CRT even for DebugGame unless the host explicitly opts in to a debug CRT.
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
set(CANE_UNREAL_STAGE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/Source/ThirdParty/CaneCore" CACHE PATH "Core SDK staging directory")
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../cane-core" "${CMAKE_CURRENT_BINARY_DIR}/core")
install(TARGETS cane-core ARCHIVE DESTINATION "${CANE_UNREAL_STAGE_DIR}/Lib/Win64")
install(DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../cane-core/include/" DESTINATION "${CANE_UNREAL_STAGE_DIR}/Include")
option(CANE_UNREAL_BUILD_TESTS "Build local portable adapter checks (not Unreal acceptance)" OFF)
if(CANE_UNREAL_BUILD_TESTS)
  enable_testing()
  add_subdirectory(tests)
endif()
